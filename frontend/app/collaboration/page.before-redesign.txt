"use client";

import { useState, useEffect } from "react";
import Link from "next/link";
import ReactMarkdown from "react-markdown";
import remarkGfm from "remark-gfm";
import { createClient } from "@/lib/supabase/client";

const markdownTest = `

- Answers questions
- Explains mistakes
- Provides practice

`;

type Contribution = { turn: number; round: number; model: string; response: string };
type ModelUsage = { turn: number; requested_provider: string; provider: string; model: string };
type CollaborationResult = {
  status: string;
  task: string;
  total_turns: number;
  discussion: Contribution[];
  model_usage: ModelUsage[];
  final_answer?: string;
  task_id?: string;
  requests_remaining?: number;
};
type HistoryTask = {
  id: string;
  task: string;
  status: string;
  selected_models: string[];
  max_rounds: number;
  current_turn: number;
  final_answer: string | null;
  created_at: string;
  updated_at: string;
  completed_at: string | null;
};

type SavedTurn = {
  id: string;
  task_id: string;
  turn_number: number;
  round_number: number;
  requested_provider: string;
  actual_provider: string;
  model_id: string;
  response: string;
  created_at: string;
};

type HistoryDetails = {
  task: HistoryTask;
  turns: SavedTurn[];
};

const API_BASE = (process.env.NEXT_PUBLIC_API_URL || "http://localhost:8000").replace(/\/$/, "");
const MODEL_NAMES: Record<string, string> = {
  gemini: "Gemini 3.5 Flash-Lite",
  groq: "Groq GPT-OSS 20B",
  openrouter: "OpenRouter (fallback)",
};

export default function CollaborationPage() {
  const [task, setTask] = useState("");
  const [result, setResult] = useState<CollaborationResult | null>(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");
  const [history, setHistory] = useState<HistoryTask[]>([]);
  const [historyLoading, setHistoryLoading] = useState(false);
  const [historyError, setHistoryError] = useState("");
  const [openingTask, setOpeningTask] = useState<string | null>(null);
  const [activeTaskId, setActiveTaskId] = useState<string | null>(null);
  const [requestsRemaining, setRequestsRemaining] = useState<number | null>(null);

async function loadHistory() {
  setHistoryLoading(true);
  setHistoryError("");

  try {
    const supabase = createClient();

    const {
      data: { session },
      error: sessionError,
    } = await supabase.auth.getSession();

    if (sessionError || !session?.access_token) {
      throw new Error("Please log in to view your history.");
    }

    const response = await fetch(`${API_BASE}/ai/history`, {
      method: "GET",
      headers: {
        Authorization: `Bearer ${session.access_token}`,
      },
      cache: "no-store",
    });

    if (!response.ok) {
      throw new Error(`Unable to load history (${response.status}).`);
    }

    const data = await response.json();

    if (!Array.isArray(data.tasks)) {
      throw new Error("Invalid history response.");
    }

    setHistory(data.tasks);
  } catch (err) {
    setHistoryError(
      err instanceof Error ? err.message : "Unable to load history."
    );
  } finally {
    setHistoryLoading(false);
  }
}

useEffect(() => {
  void loadHistory();
}, []);


async function openSavedTask(taskId: string) {
  if (loading || openingTask) return;

  setOpeningTask(taskId);
  setError("");

  try {
    const supabase = createClient();

    const {
      data: { session },
      error: sessionError,
    } = await supabase.auth.getSession();

    if (sessionError || !session?.access_token) {
      throw new Error("Please log in to open saved collaborations.");
    }

    const response = await fetch(
      `${API_BASE}/ai/history/${taskId}`,
      {
        method: "GET",
        headers: {
          Authorization: `Bearer ${session.access_token}`,
        },
        cache: "no-store",
      }
    );

    if (!response.ok) {
      throw new Error(
        `Unable to open collaboration (${response.status}).`
      );
    }

    const data: HistoryDetails = await response.json();

    if (!data.task || !Array.isArray(data.turns)) {
      throw new Error("Invalid collaboration history response.");
    }

    const savedResult: CollaborationResult = {
      status: data.task.status,
      task: data.task.task,
      task_id: data.task.id,
      total_turns: data.turns.length,

      discussion: data.turns.map((turn) => ({
        turn: turn.turn_number,
        round: turn.round_number,
        model: turn.requested_provider,
        response: turn.response,
      })),

      model_usage: data.turns.map((turn) => ({
        turn: turn.turn_number,
        requested_provider: turn.requested_provider,
        provider: turn.actual_provider,
        model: turn.model_id,
      })),

      final_answer: data.task.final_answer ?? undefined,
    };

    setTask(data.task.task);
    setResult(savedResult);
    setActiveTaskId(data.task.id);

  } catch (err) {
    setError(
      err instanceof Error
        ? err.message
        : "Unable to open saved collaboration."
    );
  } finally {
    setOpeningTask(null);
  }
}

function startNewCollaboration() {
  if (loading || openingTask) return;

  setTask("");
  setResult(null);
  setError("");
  setActiveTaskId(null);
}

  async function handleGenerate() {
    if (!task.trim() || loading || openingTask) return;
    setLoading(true);
    setError("");
    setResult(null);
    setActiveTaskId(null);

    try {
  const supabase = createClient();

  const {
    data: { session },
    error: sessionError,
  } = await supabase.auth.getSession();

  if (sessionError || !session?.access_token) {
    throw new Error("Please log in before starting collaboration.");
  }

  const response = await fetch(`${API_BASE}/ai/collaborate`, {
        method: "POST",
        headers: {
  "Content-Type": "application/json",
  Authorization: `Bearer ${session.access_token}`,
},
        body: JSON.stringify({ task: task.trim() }),
      });

      const data: unknown = await response.json();
      if (!response.ok) {
        const detail = data && typeof data === "object" && "detail" in data
          ? String(data.detail)
          : `HTTP ${response.status}`;
        throw new Error(`Collaboration failed: ${detail}`);
      }
      if (!data || typeof data !== "object") throw new Error("Invalid backend response.");
      const parsed = data as CollaborationResult;
      if (parsed.status !== "completed" || !Array.isArray(parsed.discussion) ||
          !Array.isArray(parsed.model_usage) || parsed.discussion.length === 0) {
        throw new Error("The backend returned an incomplete collaboration.");
      }
      setResult(parsed);
setActiveTaskId(parsed.task_id ?? null);

if (typeof parsed.requests_remaining === "number") {
  setRequestsRemaining(parsed.requests_remaining);
}

await loadHistory();
    } catch (err) {
      setError(err instanceof Error ? err.message : "Unexpected error.");
    } finally {
      setLoading(false);
    }
  }

  return (
    <main className="min-h-screen bg-slate-50 px-4 py-8 text-slate-900 sm:px-8">
    <div className="mx-auto grid max-w-7xl items-start gap-6 lg:grid-cols-[280px_minmax(0,1fr)]">

<aside className="rounded-2xl border border-slate-200 bg-white p-4 shadow-sm lg:sticky lg:top-8">
  <h2 className="text-lg font-bold">
    Collaboration History
  </h2>

  <button
    type="button"
    onClick={startNewCollaboration}
    disabled={loading || !!openingTask}
    className="mt-4 w-full rounded-xl bg-indigo-600 px-4 py-3 text-sm font-semibold text-white hover:bg-indigo-700 disabled:opacity-50"
  >
    + New Collaboration
  </button>

  {requestsRemaining !== null && (
    <div className="mt-4 rounded-xl bg-indigo-50 p-3">
      <p className="text-xs font-medium text-indigo-700">
        Daily requests remaining
      </p>
      <p className="mt-1 text-xl font-bold text-indigo-900">
        {requestsRemaining} / 10
      </p>
    </div>
  )}

  <div className="mt-5 flex items-center justify-between">
    <h3 className="text-sm font-semibold text-slate-600">
      Recent Collaborations
    </h3>

    <button
      type="button"
      onClick={() => void loadHistory()}
      disabled={historyLoading || loading || !!openingTask}
      className="text-xs font-medium text-indigo-600 hover:underline disabled:opacity-50"
    >
      Refresh
    </button>
  </div>

  {historyLoading && (
    <p className="mt-3 text-sm text-slate-500">
      Loading history...
    </p>
  )}

  {historyError && (
    <p className="mt-3 text-sm text-red-600">
      {historyError}
    </p>
  )}

  {!historyLoading && !historyError && history.length === 0 && (
    <p className="mt-3 text-sm text-slate-500">
      No saved collaborations yet.
    </p>
  )}

  <div className="mt-3 max-h-[60vh] space-y-2 overflow-y-auto">
    {history.map((item) => (
      <button
        key={item.id}
        type="button"
        onClick={() => void openSavedTask(item.id)}
        disabled={loading || !!openingTask}
        className={`w-full rounded-xl border p-3 text-left transition disabled:opacity-50 ${
          activeTaskId === item.id
            ? "border-indigo-300 bg-indigo-50"
            : "border-slate-200 hover:bg-slate-50"
        }`}
      >
        <p className="truncate text-sm font-medium text-slate-800">
          {item.task}
        </p>

        <div className="mt-2 flex items-center justify-between gap-2">
          <span
            className={`text-xs capitalize ${
              item.status === "completed"
                ? "text-emerald-600"
                : "text-amber-600"
            }`}
          >
            {openingTask === item.id
              ? "Opening..."
              : item.status}
          </span>

          <span className="text-xs text-slate-400">
            {new Date(item.created_at).toLocaleDateString()}
          </span>
        </div>
      </button>
    ))}
  </div>
</aside>

<div className="min-w-0">

        <Link href="/dashboard" className="text-sm font-medium text-indigo-600 hover:underline">
          ← Back to Dashboard
        </Link>

        <header className="mb-8 mt-6">
          <span className="rounded-full bg-indigo-50 px-3 py-1 text-xs font-semibold text-indigo-700">
            MULTI-AGENT AI PLATFORM
          </span>
          <h1 className="mt-4 text-3xl font-bold sm:text-4xl">AI Collaboration</h1>
          <p className="mt-2 text-slate-600">Gemini and Groq collaborate on one task, with automatic fallback if necessary.</p>
        </header>

        <section className="rounded-3xl border border-slate-200 bg-white p-5 shadow-sm sm:p-7">
          <label htmlFor="task" className="mb-3 block text-lg font-semibold">Enter Your Task</label>
          <textarea
            id="task" value={task} onChange={(e) => setTask(e.target.value)}
            rows={4} maxLength={5000} disabled={loading}
            placeholder="Explain a topic, solve a problem, or plan a project..."
            className="w-full resize-y rounded-2xl border border-slate-200 bg-slate-50 p-4 outline-none focus:border-indigo-500 focus:ring-2 focus:ring-indigo-100"
          />
          <div className="mt-4 flex flex-wrap items-center gap-3">
            <button type="button" onClick={handleGenerate} disabled={loading || !task.trim()}
              className="rounded-xl bg-indigo-600 px-6 py-3 text-sm font-semibold text-white hover:bg-indigo-700 disabled:cursor-not-allowed disabled:opacity-50">
              {loading ? "Collaborating..." : "Start Collaboration"}
            </button>
            <span className="text-sm text-slate-500">2 models · 2 rounds · 4 turns + synthesis</span>
          </div>
          {loading && <p role="status" className="mt-4 text-sm text-indigo-600">Waiting for all AI turns and final synthesis. This can take a few minutes.</p>}
          {error && <p role="alert" className="mt-4 whitespace-pre-wrap text-sm text-red-600">{error}</p>}
        </section>

        <section className="mt-8">
          <h2 className="mb-4 text-xl font-bold">Configured AI Models</h2>
          <div className="grid gap-4 sm:grid-cols-2">
            {["gemini", "groq"].map((provider) => (
              <div key={provider} className="rounded-2xl border border-slate-200 bg-white p-5 shadow-sm">
                <h3 className="font-semibold">{MODEL_NAMES[provider]}</h3>
                <p className="mt-1 text-sm text-slate-500">Scheduled participant; actual provider is reported below.</p>
              </div>
            ))}
          </div>
        </section>

        {result && (
          <>
            <section className="mt-8 rounded-3xl border border-slate-200 bg-white p-5 shadow-sm sm:p-7">
              <div className="flex flex-wrap items-center justify-between gap-3">
                <h2 className="text-xl font-bold">Collaboration Discussion</h2>
                <span className="rounded-full bg-emerald-50 px-3 py-1 text-xs font-semibold text-emerald-700">
                  {result.total_turns} turns completed
                </span>
              </div>
              <div className="mt-5 space-y-4">
                {result.discussion.map((turn, index) => {
                  const usage = result.model_usage.find((item) => item.turn === turn.turn);
                  const actual = usage?.provider || "unknown";
                  const fallback = Boolean(usage && usage.requested_provider !== usage.provider);
                  return (
                   <article
                    key={`${turn.turn}-${index}`}
                      className="rounded-2xl border border-slate-200 bg-slate-50 p-5"
                    >
                  <div className="flex flex-wrap items-center justify-between gap-2">
                  <h3 className="font-semibold">
                    Turn {turn.turn} · Round {turn.round}
                  </h3>
                  <span className="text-sm font-medium text-indigo-700">
                    {MODEL_NAMES[actual] || actual}
                </span>
            </div>

  <p className="mt-1 break-words text-xs text-slate-500">
    Requested: {turn.model} · Actual model: {usage?.model || "not reported"}
  </p>

  {fallback && (
    <p className="mt-1 text-xs font-semibold text-amber-700">
      Fallback provider used
    </p>
  )}

  <div className="markdown-content mt-4 text-sm text-slate-700">
    <ReactMarkdown remarkPlugins={[remarkGfm]}>
      {turn.response}
    </ReactMarkdown>
  </div>
</article>
                  );
                })}
              </div>
            </section>

            <section className="mt-8 rounded-3xl border border-slate-200 bg-white p-5 shadow-sm sm:p-7">
              <h2 className="text-xl font-bold">Final Collaborative Answer</h2>
              {result.final_answer?.trim() ? (
              <div className="markdown-content mt-5 text-slate-700">
              <ReactMarkdown remarkPlugins={[remarkGfm]}>
                {result.final_answer}
              </ReactMarkdown>
              </div>
              ) : (
  <p className="mt-4 text-amber-700">
    The backend returned the discussion, but no final synthesis was included.
  </p>
)}
            </section>
          </>
                )}
        </div>
      </div>
    </main>
  );
}
